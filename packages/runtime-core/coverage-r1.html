type LogMode string

const (
        LogModeNone            LogMode = "none"
        LogModeTransitionsOnly LogMode = "transitions_only"
        LogModeFull            LogMode = "full"
        LogModeReplay          LogMode = "replay"

type Type string

const (
        TypeSpinChainStart       Type = "SPIN_CHAIN_START"
        TypeSpinChainEnd         Type = "SPIN_CHAIN_END"
        TypeFramePush            Type = "FRAME_PUSH"
        TypeFramePop             Type = "FRAME_POP"
        TypeStateEnter           Type = "STATE_ENTER"
        TypeStateReenter         Type = "STATE_REENTER"
        TypeGridResult           Type = "GRID_RESULT"
        TypeWinEvaluated         Type = "WIN_EVALUATED"
        TypeTriggerEmitted       Type = "TRIGGER_EMITTED"
        TypeTransitionTaken      Type = "TRANSITION_TAKEN"
        TypeMultiplierApplied    Type = "MULTIPLIER_APPLIED"
        TypeExposureCapApplied   Type = "EXPOSURE_CAP_APPLIED"
        TypeFeatureCounterUpdate Type = "FEATURE_COUNTER_UPDATE"
        TypeRuntimeLimitHit      Type = "RUNTIME_LIMIT_HIT"
        TypeRuntimeFault         Type = "RUNTIME_FAULT"
        TypeRNGDraw              Type = "RNG_DRAW"

type Correlation struct {
        SimulationRunID        string `json:"simulationRunId"`
        SpinChainIndex         int64  `json:"spinChainIndex"`
        GameVersionContentHash string `json:"gameVersionContentHash"`
        EngineVersion          string `json:"engineVersion"`

type Context struct {
        StackDepth int    `json:"stackDepth"`
        MachineID  string `json:"machineId,omitempty"`

type Event struct {
        EventVersion string         `json:"eventVersion"`
        Sequence     int            `json:"sequence"`
        EventType    Type           `json:"eventType"`
        TimestampUs  int64          `json:"timestampUs,omitempty"`
        Correlation  Correlation    `json:"correlation"`
        Context      Context        `json:"context"`

}

type replayUnit struct {
        EventVersion string         `json:"eventVersion"`
        Sequence     int            `json:"sequence"`
        EventType    Type           `json:"eventType"`
        Correlation  Correlation    `json:"correlation"`
        Context      Context        `json:"context"`
        Payload      map[string]any `json:"payload"`
}

func (e Event) ReplayUnit() replayUnit {
        return replayUnit{
                EventVersion: e.EventVersion,
                Sequence:     e.Sequence,
                EventType:    e.EventType,
                Correlation:  e.Correlation,
                Context:      e.Context,
                Payload:      e.Payload,
        }
}

type Logger struct {
        mode        LogMode
        correlation Correlation
        sequence    int
        events      []Event
}

func NewLogger(mode LogMode, correlation Correlation) *Logger {
        if mode == "" {
                mode = LogModeNone
        }
        return &Logger{mode: mode, correlation: correlation}
}

func (l *Logger) Mode() LogMode {
        return l.mode
}

func (l *Logger) Append(event Event) {
        if !ShouldEmit(event.EventType, l.mode) {
                return
        }
        event.EventVersion = EventVersion
        event.Sequence = l.sequence
        event.Correlation = l.correlation
        l.sequence++
        l.events = append(l.events, event)
}

func (l *Logger) Events() []Event {
        out := make([]Event, len(l.events))
        copy(out, l.events)
        return out
}

func (l *Logger) Hash() string {
        if len(l.events) == 0 {
                return ""
        }
        h := sha256.New()
        for _, event := range l.events {
                line, err := json.Marshal(event.ReplayUnit())
                if err != nil {
                        continue
                }
                h.Write(line)
                h.Write([]byte{'\n'})
        }
        return hex.EncodeToString(h.Sum(nil))
}

func ShouldEmit(eventType Type, mode LogMode) bool {
        switch mode {
        case LogModeNone:
                return false
        case LogModeTransitionsOnly:
                return isTransitionModeEvent(eventType)
        case LogModeFull:
                return isMVPEvent(eventType)
        case LogModeReplay:
                return isMVPEvent(eventType) || eventType == TypeRNGDraw || eventType == TypeGuardEvaluated
        default:
                return false
        }
}

func isTransitionModeEvent(t Type) bool {
        switch t {
        case TypeSpinChainStart, TypeSpinChainEnd, TypeFramePush, TypeFramePop,
                TypeStateEnter, TypeTransitionTaken:
                return true
        default:
                return false
        }
}

func isMVPEvent(t Type) bool {
        switch t {
        case TypeSpinChainStart, TypeSpinChainEnd, TypeFramePush, TypeFramePop,
                TypeStateEnter, TypeStateReenter, TypeGridResult, TypeWinEvaluated,
                TypeTriggerEmitted, TypeTransitionTaken, TypeMultiplierApplied,
                TypeExposureCapApplied, TypeFeatureCounterUpdate, TypeRuntimeLimitHit,
                TypeRuntimeFault:
                return true
        default:
                return false
        }
}

package rng

// PCG64 is a permuted congruential generator (64-bit output).
// Reference: O'Neill, PCG: A Family of Simple Fast Space-Efficient Statistically Good Algorithms for Random Number Generation.
type PCG64 struct {
        state uint64
        inc   uint64
}

func newPCG64(initState, initSeq uint64) *PCG64 {
        if initSeq == 0 {
                initSeq = 1
        }
        p := &PCG64{inc: (initSeq << 1) | 1}
        p.state = 0
        p.NextUint64()
        p.state += initState
        p.NextUint64()
        return p
}

func (p *PCG64) NextUint64() uint64 {
        oldState := p.state
        p.state = oldState*6364136223846793005 + p.inc
        xorshifted := uint32(((oldState >> 18) ^ oldState) >> 27)
        rot := oldState >> 59
        return (uint64(xorshifted) >> rot) | (uint64(xorshifted) << ((-int(rot)) & 31))
}

package rng

import (
        "crypto/hmac"
        "crypto/sha256"
        "encoding/binary"
        "fmt"
)

type Algorithm string

const AlgorithmPCG64 Algorithm = "PCG64"

type DerivationPolicy string

const PolicyV1HashShard DerivationPolicy = "v1-hash-shard"

type Config struct {
        Algorithm Algorithm
        Policy    DerivationPolicy
}

type Service struct {
        algorithm Algorithm
        policy    DerivationPolicy
}

func NewService(cfg Config) (*Service, error) {
        if cfg.Algorithm == "" {
                cfg.Algorithm = AlgorithmPCG64
        }
        if cfg.Policy == "" {
                cfg.Policy = PolicyV1HashShard
        }
        if cfg.Algorithm != AlgorithmPCG64 {
                return nil, fmt.Errorf("unsupported rng algorithm %q", cfg.Algorithm)
        }
        if cfg.Policy != PolicyV1HashShard {
                return nil, fmt.Errorf("unsupported seed derivation policy %q", cfg.Policy)
        }
        return &Service{algorithm: cfg.Algorithm, policy: cfg.Policy}, nil
}

type ChainScope struct {
        chainSeed []byte
}

func (s *Service) NewChainScope(masterSeed []byte, shardIndex int, simulationRunID string, spinChainIndex int64) *ChainScope {
        shardSeed := deriveSeed(masterSeed, encodeInt(shardIndex), []byte(simulationRunID))
        chainSeed := deriveSeed(shardSeed, encodeInt64(spinChainIndex))
        return &ChainScope{chainSeed: chainSeed}
}

func (c *ChainScope) ChainSeed() []byte {
        out := make([]byte, len(c.chainSeed))
        copy(out, c.chainSeed)
        return out
}

type FrameStream struct {
        frameStreamSeed []byte
}

func (c *ChainScope) FrameStream(frameID string) *FrameStream {
        seed := deriveSeed(c.chainSeed, []byte(frameID))
        return &FrameStream{frameStreamSeed: seed}
}

func (f *FrameStream) StepRNG(stepIndex int) *RNG {
        label := append([]byte("step:"), encodeInt(stepIndex)...)
        seed := deriveSeed(f.frameStreamSeed, label)
        return newRNG(seed)
}

func (f *FrameStream) MultiplierRNG(multiplierRefID string, applyIndex int) *RNG {
        label := append([]byte("mult:"), []byte(multiplierRefID)...)
        label = append(label, ':')
        label = append(label, encodeInt(applyIndex)...)
        seed := deriveSeed(f.frameStreamSeed, label)
        return newRNG(seed)
}

func deriveSeed(key []byte, parts ...[]byte) []byte {
        mac := hmac.New(sha256.New, key)
        for _, part := range parts {
                mac.Write(part)
        }
        sum := mac.Sum(nil)
        return sum
}

func encodeInt(v int) []byte {
        buf := make([]byte, 8)
        binary.BigEndian.PutUint64(buf, uint64(v))
        return buf
}

func encodeInt64(v int64) []byte {
        buf := make([]byte, 8)
        binary.BigEndian.PutUint64(buf, uint64(v))
        return buf
}

func seedToPCG64State(seed []byte) (state, seq uint64) {
        if len(seed) < 16 {
                padded := make([]byte, 16)
                copy(padded, seed)
                seed = padded
        }
        state = binary.BigEndian.Uint64(seed[:8])
        seq = binary.BigEndian.Uint64(seed[8:16])
        if seq == 0 {
                seq = 54
        }
        return state, seq
}

func newRNG(seed []byte) *RNG {
        state, seq := seedToPCG64State(seed)
        return &RNG{pcg: newPCG64(state, seq)}
}

// RNG implements the plugin-facing draw interface.
type RNG struct {
        pcg    *PCG64
        drawID int
}

func (r *RNG) NextUint64() uint64 {
        r.drawID++
        return r.pcg.NextUint64()
}

func (r *RNG) NextInt(n int) int {
        if n <= 0 {
                panic(fmt.Sprintf("rng: NextInt requires n > 0, got %d", n))
        }
        if n == 1 {
                r.drawID++
                return 0
        }
        max := uint64(n)
        limit := (^uint64(0) / max) * max
        for {
                x := r.pcg.NextUint64()
                if x < limit {
                        r.drawID++
                        return int(x % max)
                }
        }
}

func (r *RNG) DrawID() int {
        return r.drawID
}

        MaxFramesPerChain      int

        MaxWallClockMsPerChain int

        EnforceCap        bool

        MaxWinMultiplier  int64

type Session struct {
        GameVersionContentHash string
        EngineVersion          string
        SpinChainIndex         int64
        Bet                    BetContext
        TotalBetMinorUnits     int64
        InheritancePolicy      string
        Limits                 EffectiveLimits
        Exposure               ExposureState
}

func New(params Params) (*Session, error) {
        if params.Game == nil {
                return nil, fmt.Errorf("game is required")
        }
        if params.EngineVersion == "" {
                return nil, fmt.Errorf("engineVersion is required")
        }
        if len(params.MasterSeed) == 0 {
                return nil, fmt.Errorf("masterSeed is required")
        }
        if params.SimulationRunID == "" {
                return nil, fmt.Errorf("simulationRunID is required")
        }
        if err := validateDeterminism(params.Game); err != nil {
                return nil, err
        }
        hash, err := contentHash(params.Game)
        if err != nil {
                return nil, err
        }
        bet, totalBet, err := resolveBet(params.Game, params.BetMinorUnits)
        if err != nil {
                return nil, err
        }
        limits := resolveLimits(params.Game)
        exposure := resolveExposure(params.Game, totalBet)
        policy := params.Game.Session.ContextInheritancePolicy
        if policy == "" {
                policy = "mvp-v1"
        }
        return &Session{
                GameVersionContentHash: hash,
                EngineVersion:          params.EngineVersion,
                SpinChainIndex:         params.SpinChainIndex,
                Bet:                    bet,
                TotalBetMinorUnits:     totalBet,
                InheritancePolicy:      policy,
                Limits:                 limits,
                Exposure:               exposure,
        }, nil
}

func validateDeterminism(game *domain.GameVersion) error {
        d := game.Determinism
        if d.RngAlgorithm != "PCG64" {
                return fmt.Errorf("unsupported determinism.rngAlgorithm %q (MVP requires PCG64)", d.RngAlgorithm)
        }
        if d.SeedDerivationPolicy != "" && d.SeedDerivationPolicy != "v1-hash-shard" {
                return fmt.Errorf("unsupported determinism.seedDerivationPolicy %q", d.SeedDerivationPolicy)
        }
        if d.MoneyRepresentation != "" && d.MoneyRepresentation != "fixed_minor_units" {
                return fmt.Errorf("unsupported determinism.moneyRepresentation %q", d.MoneyRepresentation)
        }
        return nil
}

func contentHash(game *domain.GameVersion) (string, error) {
        if game.ContentHash != "" {
                return game.ContentHash, nil
        }
        raw, err := json.Marshal(game)
        if err != nil {
                return "", fmt.Errorf("marshal game for content hash: %w", err)
        }
        sum := sha256.Sum256(raw)
        return hex.EncodeToString(sum[:]), nil
}

func resolveBet(game *domain.GameVersion, override int64) (BetContext, int64, error) {
        bet := game.BetModel
        if override > 0 {
                return BetContext{
                        LineCount:          bet.LineCount,
                        CoinValueScale:     bet.CoinValueScale,
                        DefaultCoinValue:   bet.DefaultCoinValue,
                        TotalBetMinorUnits: override,
                        BetModelType:       bet.Type,
                }, override, nil
        }
        if bet.TotalBetMinorUnits > 0 {
                return BetContext{
                        LineCount:          bet.LineCount,
                        CoinValueScale:     bet.CoinValueScale,
                        DefaultCoinValue:   bet.DefaultCoinValue,
                        TotalBetMinorUnits: int64(bet.TotalBetMinorUnits),
                        BetModelType:       bet.Type,
                }, int64(bet.TotalBetMinorUnits), nil
        }
        lines := bet.LineCount
        if lines <= 0 {
                lines = 1
        }
        scale := bet.CoinValueScale
        if scale <= 0 {
                scale = 1
        }
        coin := bet.DefaultCoinValue
        if coin <= 0 {
                coin = 1
        }
        total := int64(lines * coin * scale)
        return BetContext{
                LineCount:          bet.LineCount,
                CoinValueScale:     bet.CoinValueScale,
                DefaultCoinValue:   bet.DefaultCoinValue,
                TotalBetMinorUnits: total,
                BetModelType:       bet.Type,
        }, total, nil
}

func resolveLimits(game *domain.GameVersion) EffectiveLimits {
        defaults := platformLimits()
        limits := defaults
        if game.Session.RuntimeLimits != nil {
                rl := game.Session.RuntimeLimits
                if rl.MaxStepsPerSpinChain > 0 {
                        limits.MaxStepsPerSpinChain = rl.MaxStepsPerSpinChain
                }
                if rl.MaxFramesPerChain > 0 {
                        limits.MaxFramesPerChain = rl.MaxFramesPerChain
                }
                if rl.MaxTransitionsPerStep > 0 {
                        limits.MaxTransitionsPerStep = rl.MaxTransitionsPerStep
                }
                if rl.MaxRespinsPerFeature > 0 {
                        limits.MaxRespinsPerFeature = rl.MaxRespinsPerFeature
                }
                if rl.MaxEventsPerChain > 0 {
                        limits.MaxEventsPerChain = rl.MaxEventsPerChain
                }
                if rl.MaxWallClockMsPerChain > 0 {
                        limits.MaxWallClockMsPerChain = rl.MaxWallClockMsPerChain
                }
        }
        maxStack := game.Session.MaxStackDepth
        if maxStack <= 0 {
                maxStack = 8
        }
        if limits.MaxFramesPerChain <= 0 || limits.MaxFramesPerChain > maxStack {
                limits.MaxFramesPerChain = maxStack
        }
        return limits
}

func platformLimits() EffectiveLimits {
        return EffectiveLimits{
                MaxStepsPerSpinChain:   500,
                MaxFramesPerChain:      8,
                MaxTransitionsPerStep:  8,
                MaxRespinsPerFeature:   100,
                MaxEventsPerChain:      10000,
                MaxWallClockMsPerChain: 0,
        }
}

func resolveExposure(game *domain.GameVersion, totalBet int64) ExposureState {
        exp := game.Exposure
        state := ExposureState{
                EnforceCap:       exp.EnforceCap,
                CapScope:         exp.CapScope,
                MaxWinMinorUnits: int64(exp.MaxWinMinorUnits),
                MaxWinMultiplier: int64(exp.MaxWinMultiplier),
        }
        cap := state.MaxWinMinorUnits
        if exp.MaxWinMultiplier > 0 && totalBet > 0 {
                multCap := int64(exp.MaxWinMultiplier) * totalBet
                if cap == 0 || multCap < cap {
                        cap = multCap
                }
        }
        state.EffectiveCapMinor = cap
        return state
}

type EventContext = ie.Context

// NewEventLogger constructs an event logger for a spin chain.
func NewEventLogger(mode EventLogMode, correlation EventCorrelation) *EventLogger {
        return &EventLogger{inner: ie.NewLogger(ie.LogMode(mode), correlation)}

// EventLogger records runtime events with monotonic sequencing and replay hashing.
type EventLogger struct {
        inner *ie.Logger
}

// Mode returns the configured event log mode.

        return EventLogMode(l.inner.Mode())
}

        l.inner.Append(event)
}

// Events returns emitted events in sequence order.

        return l.inner.Events()
}

        return l.inner.Hash()
}

        return ie.ShouldEmit(eventType, ie.LogMode(mode))
}

        return json.Marshal(event.ReplayUnit())

                MasterSeed:      req.MasterSeed,
                SpinChainIndex:  req.SpinChainIndex,
                ShardIndex:      req.ShardIndex,
                SimulationRunID: req.SimulationRunID,
                BetMinorUnits:   req.BetMinorUnits,
                EventLogMode:    req.EventLogMode,
                RNGConfig:       rngCfg,
        }
}

// NewSessionContext constructs a session for one spin chain.
func NewSessionContext(params NewSessionParams) (*SessionContext, error) {
        inner, err := isession.New(isession.Params{
                Game:            params.Game,
                EngineVersion:   params.EngineVersion,
                MasterSeed:      params.MasterSeed,
                SpinChainIndex:  params.SpinChainIndex,
                ShardIndex:      params.ShardIndex,
                SimulationRunID: params.SimulationRunID,
                BetMinorUnits:   params.BetMinorUnits,
                EventLogMode:    isession.EventLogMode(params.EventLogMode),
                RNGConfig: isession.RNGConfig{
                        Algorithm: isession.RNGAlgorithm(params.RNGConfig.Algorithm),
                        Policy:    isession.RNGPolicy(params.RNGConfig.Policy),
                },
        })
        if err != nil {
                return nil, err
        }
        rngSvc, err := NewRNGService(params.RNGConfig)
        if err != nil {
                return nil, fmt.Errorf("rng service: %w", err)
        }
        chainScope := rngSvc.NewChainScope(params.MasterSeed, params.ShardIndex, params.SimulationRunID, params.SpinChainIndex)
        logger := NewEventLogger(params.EventLogMode, EventCorrelation{
                SimulationRunID:        params.SimulationRunID,
                SpinChainIndex:         params.SpinChainIndex,
                GameVersionContentHash: inner.GameVersionContentHash,
                EngineVersion:          params.EngineVersion,
        })
        return &SessionContext{
                GameVersionContentHash:   inner.GameVersionContentHash,
                EngineVersion:            inner.EngineVersion,
                SpinChainIndex:           inner.SpinChainIndex,
                Bet:                      mapBetContext(inner.Bet),
                TotalBetMinorUnits:       inner.TotalBetMinorUnits,
                InheritancePolicy:        inner.InheritancePolicy,
                RNG:                      rngSvc,
                ChainRNG:                 chainScope,
                Events:                   logger,
                Limits:                   mapLimits(inner.Limits),
                Exposure:                 mapExposure(inner.Exposure),
        }, nil
}

func mapBetContext(b isession.BetContext) BetContext {
        return BetContext{
                LineCount:          b.LineCount,
                CoinValueScale:     b.CoinValueScale,
                DefaultCoinValue:   b.DefaultCoinValue,
                TotalBetMinorUnits: b.TotalBetMinorUnits,
                BetModelType:       b.BetModelType,
        }
}

func mapLimits(l isession.EffectiveLimits) EffectiveLimits {
        return EffectiveLimits{
                MaxStepsPerSpinChain:   l.MaxStepsPerSpinChain,
                MaxFramesPerChain:      l.MaxFramesPerChain,
                MaxTransitionsPerStep:  l.MaxTransitionsPerStep,
                MaxRespinsPerFeature:   l.MaxRespinsPerFeature,
                MaxEventsPerChain:      l.MaxEventsPerChain,
                MaxWallClockMsPerChain: l.MaxWallClockMsPerChain,
        }
}

func mapExposure(e isession.ExposureState) ExposureState {
        return ExposureState{
                EnforceCap:        e.EnforceCap,
                EffectiveCapMinor: e.EffectiveCapMinor,
                CapScope:          e.CapScope,
                MaxWinMinorUnits:  e.MaxWinMinorUnits,
                MaxWinMultiplier:  e.MaxWinMultiplier,
        }
}
